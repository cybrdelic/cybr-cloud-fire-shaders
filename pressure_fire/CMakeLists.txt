cmake_minimum_required(VERSION 3.20)
project(cybr_pressure_fire LANGUAGES CXX)
add_executable(euler euler.cpp)
target_compile_features(euler PRIVATE cxx_std_17)
set_target_properties(euler PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(MSVC)
  target_compile_options(euler PRIVATE /O2 /EHsc)
  message(STATUS "MSVC: serial solver (its OpenMP 2.0 does not support this solver's reductions)")
else()
  target_compile_options(euler PRIVATE -O3)
  find_package(OpenMP QUIET)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(euler PRIVATE OpenMP::OpenMP_CXX)
  else()
    message(STATUS "OpenMP unavailable: building the same solver in serial mode")
  endif()
endif()
